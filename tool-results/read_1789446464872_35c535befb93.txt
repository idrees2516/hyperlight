     1→//! Central book registry: shared state between venue connectors, the publish
     2→//! loop, the depth sampler and the HTTP/WS layer.
     3→//!
     4→//! Scaling model (12 markets): every socket selects ONE market. The broadcast
     5→//! channel carries a compact `Broadcast` envelope — (kind, market, json) — so
     6→//! each socket can route events without re-parsing JSON:
     7→//! - `Book`   → forwarded only to sockets whose selection matches (full depth)
     8→//! - `Ticker` → forwarded to everyone (compact, coalesced at 2 Hz)
     9→//! - `Trades` → forwarded to everyone (batches, small)
    10→//! - `AlertSet` / `AlertFired` / `Status` → forwarded to everyone
    11→

    13→    compute_stats, consolidate, cross_mid, Alert, AlertDir, BookStats, ConsolidatedBook,
    14→    DepthSample, FeedStatus, Market, Trade, VenueBook, VenueHealth, VenueState, WireEvent,
    15→    SAMPLE_SECS, SAMPLE_WINDOW, TAPE_BACKLOG, WIRE_DEPTH,
    16→};
    17→use rust_decimal::prelude::ToPrimitive;
    18→use rust_decimal::Decimal;
    19→use serde_json::json;

    26→pub fn now_ms() -> u64 {
    27→    SystemTime::now()
    28→        .duration_since(UNIX_EPOCH)
    29→        .map(|d| d.as_millis() as u64)
    30→        .unwrap_or(0)
    31→}

    45→    /// Should a socket with selection `sel` receive an event about `market`?
    46→    pub fn wants(self, sel: Market, market: Market) -> bool {
    47→        match self {
    48→            BcKind::Book => sel == market,
    49→            _ => true,
    50→        }
    51→    }

    62→/// Live telemetry for one venue's websocket feed.
    63→pub struct VenueFeed {
    64→    status: Mutex<FeedStatus>,
    65→    total_msgs: AtomicU64,
    66→    last_msg_ms: AtomicU64,
    67→    ema_rate: Mutex<f32>,
    68→}
    69→
    70→impl VenueFeed {
    71→    fn new() -> Self {
    72→        Self {
    73→            status: Mutex::new(FeedStatus::Connecting),
    74→            total_msgs: AtomicU64::new(0),
    75→            last_msg_ms: AtomicU64::new(0),
    76→            ema_rate: Mutex::new(0.0),
    77→        }
    78→    }
    79→
    80→    pub fn set_status(&self, s: FeedStatus) {
    81→        *self.status.lock().unwrap() = s;
    82→    }
    83→
    84→    pub fn status(&self) -> FeedStatus {
    85→        *self.status.lock().unwrap()
    86→    }
    87→
    88→    pub fn record_msg(&self) {
    89→        self.total_msgs.fetch_add(1, Ordering::Relaxed);
    90→        self.last_msg_ms.store(now_ms(), Ordering::Relaxed);
    91→    }
    92→
    93→    pub fn total(&self) -> u64 {
    94→        self.total_msgs.load(Ordering::Relaxed)
    95→    }
    96→
    97→    pub fn last_msg_ms(&self) -> u64 {
    98→        self.last_msg_ms.load(Ordering::Relaxed)
    99→    }
   100→
   101→    /// Track the message rate with a light EMA. Called at 10 Hz.
   102→    fn tick_rate(&self, last_sample: &mut u64, last_ts: &mut u64) -> f32 {
   103→        let now = now_ms();
   104→        let total = self.total();
   105→        let dt = (now - *last_ts).max(1) as f32 / 1000.0;
   106→        let inst = (total.saturating_sub(*last_sample)) as f32 / dt;
   107→        *last_sample = total;
   108→        *last_ts = now;
   109→        let mut ema = self.ema_rate.lock().unwrap();
   110→        *ema = *ema * 0.85 + inst * 0.15;
   111→        *ema
   112→    }
   113→
   114→    pub fn rate(&self) -> f32 {
   115→        *self.ema_rate.lock().unwrap()
   116→    }
   117→}

   125→/// Per-market trade tape: backlog for new sockets + dedup + publish pending.
   126→#[derive(Default)]
   127→struct TapeState {
   128→    /// Newest last.
   129→    trades: VecDeque<Trade>,
   130→    seen: HashSet<String>,
   131→    seen_order: VecDeque<String>,
   132→    /// New trades since the last publish flush.
   133→    pending: Vec<Trade>,
   134→}

   136→/// The whole application state.
   137→pub struct Registry {
   138→    books: Mutex<HashMap<Market, MarketBook>>,
   139→    pub hl: VenueFeed,
   140→    pub lt: VenueFeed,
   141→    tx: broadcast::Sender<Broadcast>,
   142→    /// Number of connected sockets currently showing each market.
   143→    selections: Mutex<HashMap<Market, u32>>,
   144→    tapes: Mutex<HashMap<Market, TapeState>>,
   145→    history: Mutex<HashMap<Market, VecDeque<DepthSample>>>,
   146→    alerts: Mutex<Vec<Alert>>,
   147→    next_alert_id: AtomicU64,
   148→    pub started_ms: u64,
   149→}

   151→impl Registry {
   152→    pub fn new() -> Arc<Self> {
   153→        let (tx, _) = broadcast::channel(1024);
   154→        let mut books = HashMap::new();
   155→        for m in Market::ALL {
   156→            books.insert(
   157→                m,
   158→                MarketBook {
   159→                    hl: None,
   160→                    lt: None,
   161→                    dirty: false,
   162→                },
   163→            );
   164→        }
   165→        Arc::new(Self {
   166→            books: Mutex::new(books),
   167→            hl: VenueFeed::new(),
   168→            lt: VenueFeed::new(),
   169→            tx,
   170→            selections: Mutex::new(HashMap::new()),
   171→            tapes: Mutex::new(HashMap::new()),
   172→            history: Mutex::new(HashMap::new()),
   173→            alerts: Mutex::new(Vec::new()),
   174→            next_alert_id: AtomicU64::new(1),
   175→            started_ms: now_ms(),
   176→        })
   177→    }

   191→    pub fn select_market(&self, m: Market) {
   192→        *self.selections.lock().unwrap().entry(m).or_insert(0) += 1;
   193→    }

   195→    pub fn deselect_market(&self, m: Market) {
   196→        let mut sel = self.selections.lock().unwrap();
   197→        if let Some(c) = sel.get_mut(&m) {
   198→            *c = c.saturating_sub(1);
   199→            if *c == 0 {
   200→                sel.remove(&m);
   201→            }
   202→        }
   203→    }

   207→    pub fn replace_hl(
   208→        &self,
   209→        market: Market,
   210→        bids: Vec<(Decimal, Decimal, Option<u32>)>,
   211→        asks: Vec<(Decimal, Decimal, Option<u32>)>,

   213→        let mut books = self.books.lock().unwrap();

   230→            mb.lt
   231→                .get_or_insert_with(VenueState::default)

   233→            mb.dirty = true;

   237→    pub fn apply_lt_side(
   238→        &self,
   239→        market: Market,
   240→        side: ob_core::Side,

   263→    /// Push venue trades into the tape (dedup by venue id, keep a backlog).
   264→    pub fn push_trades(&self, market: Market, trades: Vec<Trade>) {
   265→        let mut tapes = self.tapes.lock().unwrap();
   266→        let st = tapes.entry(market).or_default();
   267→        for t in trades {
   268→            if st.seen.contains(&t.id) {
   269→                continue;
   270→            }
   271→            st.seen.insert(t.id.clone());
   272→            st.seen_order.push_back(t.id.clone());
   273→            // Bound the dedup set to ~2x the backlog (evict in arrival order).
   274→            while st.seen_order.len() > 2 * TAPE_BACKLOG {
   275→                if let Some(old) = st.seen_order.pop_front() {
   276→                    st.seen.remove(&old);
   277→                }
   278→            }
   279→            st.trades.push_back(t.clone());
   280→            while st.trades.len() > TAPE_BACKLOG {
   281→                st.trades.pop_front();
   282→            }
   283→            st.pending.push(t);
   284→        }
   285→    }
   286→
   287→    /// Drain trades pending publication (called by the publish loop at 10 Hz).
   288→    fn take_pending(&self) -> Vec<(Market, Vec<Trade>)> {
   289→        let mut tapes = self.tapes.lock().unwrap();
   290→        tapes
   291→            .iter_mut()
   292→            .filter(|(_, st)| !st.pending.is_empty())
   293→            .map(|(m, st)| (*m, std::mem::take(&mut st.pending)))
   294→            .collect()
   295→    }

   297→    /// Recent trades for a market, oldest first (for new sockets).
   298→    pub fn tape_backlog(&self, market: Market) -> Vec<Trade> {
   299→        self.tapes
   300→            .lock()
   301→            .unwrap()
   302→            .get(&market)
   303→            .map(|st| st.trades.iter().cloned().collect())
   304→            .unwrap_or_default()
   305→    }

   309→    /// Sample depth bands for every market with live book data. Called every
   310→    /// `SAMPLE_SECS` by a dedicated task; keeps a `SAMPLE_WINDOW` ring buffer.
   311→    pub fn sample_all(&self) {
   312→        let now = now_ms();
   313→        // Band thresholds: 0.1%, 0.5%, 1%, 2%.
   314→        let ths = [
   315→            Decimal::new(1, 3),
   316→            Decimal::new(5, 3),
   317→            Decimal::new(1, 2),
   318→            Decimal::new(2, 2),
   319→        ];
   320→        let mut samples: Vec<(Market, DepthSample)> = Vec::new();
   321→        {
   322→            let books = self.books.lock().unwrap();
   323→            for (m, mb) in books.iter() {
   324→                let (hl, lt) = (mb.hl.as_ref(), mb.lt.as_ref());
   325→                if hl.is_none() && lt.is_none() {
   326→                    continue;
   327→                }
   328→                let Some(mid) = cross_mid(hl, lt) else { continue };
   329→
   330→                let mut bands = [Decimal::ZERO; 4];
   331→                let mut asks = [Decimal::ZERO; 4];
   332→                for st in [hl, lt].into_iter().flatten() {
   333→                    let b = st.band_notionals(mid, &ths, ob_core::Side::Bid);
   334→                    let a = st.band_notionals(mid, &ths, ob_core::Side::Ask);
   335→                    for i in 0..4 {
   336→                        bands[i] += b[i];
   337→                        asks[i] += a[i];
   338→                    }
   339→                }
   340→
   341→                // Spread over the cross-venue touch.
   342→                let bb = [hl.and_then(|b| b.best_bid()), lt.and_then(|b| b.best_bid())]
   343→                    .into_iter()
   344→                    .flatten()
   345→                    .max_by(|x, y| x.0.cmp(&y.0));
   346→                let ba = [hl.and_then(|b| b.best_ask()), lt.and_then(|b| b.best_ask())]
   347→                    .into_iter()
   348→                    .flatten()
   349→                    .min_by(|x, y| x.0.cmp(&y.0));
   350→                let sb = match (bb, ba) {
   351→                    (Some(bb), Some(ba)) => {
   352→                        if mid.is_zero() {
   353→                            Decimal::ZERO
   354→                        } else {
   355→                            (ba.0 - bb.0) / mid * Decimal::from(10_000u64)
   356→                        }
   357→                    }
   358→                    _ => Decimal::ZERO,
   359→                };
   360→
   361→                let total = bands[1] + asks[1];
   362→                let imb = if total.is_zero() {
   363→                    Decimal::ZERO
   364→                } else {
   365→                    (bands[1] - asks[1]) / total
   366→                };
   367→
   368→                let f = |d: Decimal| d.to_f64().unwrap_or(0.0);
   369→                samples.push((
   370→                    *m,
   371→                    DepthSample {
   372→                        t: now,
   373→                        mid: f(mid),
   374→                        sb: f(sb.round_dp(2)),
   375→                        b1: f(bands[0]),
   376→                        a1: f(asks[0]),
   377→                        b5: f(bands[1]),
   378→                        a5: f(asks[1]),
   379→                        b10: f(bands[2]),
   380→                        a10: f(asks[2]),
   381→                        b20: f(bands[3]),
   382→                        a20: f(asks[3]),
   383→                        im: f(imb.round_dp(4)),
   384→                    },
   385→                ));
   386→            }
   387→        }
   388→        if samples.is_empty() {
   389→            return;
   390→        }
   391→        let mut hist = self.history.lock().unwrap();
   392→        for (m, s) in samples {
   393→            let q = hist.entry(m).or_default();
   394→            // Skip duplicate timestamps (same sample window).
   395→            if q.back().map(|b| b.t == s.t).unwrap_or(false) {
   396→                continue;
   397→            }
   398→            q.push_back(s);
   399→            while q.len() > SAMPLE_WINDOW {
   400→                q.pop_front();
   401→            }
   402→        }
   403→    }

   427→    /// Create an alert (validates the price parses as a positive decimal).
   428→    pub fn create_alert(&self, market: Market, dir: AlertDir, price: String) -> Result<Alert, String> {
   429→        let th: Decimal = price
   430→            .parse()
   431→            .map_err(|_| format!("price {price:?} is not a valid decimal"))?;
   432→        if !th.is_sign_positive() {
   433→            return Err("price must be positive".into());
   434→        }
   435→        let alert = Alert {
   436→            id: self.next_alert_id.fetch_add(1, Ordering::Relaxed),
   437→            market,
   438→            dir,
   439→            price: th.normalize().to_string(),
   440→            created_ms: now_ms(),
   441→            triggered_ms: None,
   442→            triggered_px: None,
   443→        };
   444→        let mut alerts = self.alerts.lock().unwrap();
   445→        alerts.push(alert.clone());
   446→        drop(alerts);
   447→        self.broadcast(
   448→            BcKind::AlertSet,
   449→            alert.market,
   450→            &WireEvent::AlertSet {
   451→                alerts: self.alerts_snapshot(),
   452→            },
   453→        );
   454→        Ok(alert)
   455→    }
   456→
   457→    pub fn delete_alert(&self, id: u64) -> bool {
   458→        let removed = {
   459→            let mut alerts = self.alerts.lock().unwrap();
   460→            let before = alerts.len();
   461→            alerts.retain(|a| a.id != id);
   462→            before != alerts.len()
   463→        };
   464→        if removed {
   465→            self.broadcast(
   466→                BcKind::AlertSet,
   467→                Market::Eth,
   468→                &WireEvent::AlertSet {
   469→                    alerts: self.alerts_snapshot(),
   470→                },
   471→            );
   472→        }
   473→        removed
   474→    }

   476→    pub fn alerts_snapshot(&self) -> Vec<Alert> {
   477→        self.alerts.lock().unwrap().clone()
   478→    }

   480→    /// Check every active alert against the live cross-venue mid. Returns the
   481→    /// alerts that fired during this pass (already mutated in place).
   482→    fn check_alerts(&self) -> Vec<Alert> {
   483→        let now = now_ms();
   484→        let mut fired = Vec::new();
   485→        {
   486→            let books = self.books.lock().unwrap();
   487→            let mut alerts = self.alerts.lock().unwrap();
   488→            for a in alerts.iter_mut() {
   489→                if a.triggered_ms.is_some() {
   490→                    continue;
   491→                }
   492→                let Ok(th) = a.price.parse::<Decimal>() else {
   493→                    continue;
   494→                };
   495→                let Some(mb) = books.get(&a.market) else {
   496→                    continue;
   497→                };
   498→                let Some(mid) = cross_mid(mb.hl.as_ref(), mb.lt.as_ref()) else {
   499→                    continue;
   500→                };
   501→                let hit = match a.dir {
   502→                    AlertDir::Above => mid >= th,
   503→                    AlertDir::Below => mid <= th,
   504→                };
   505→                if hit {
   506→                    a.triggered_ms = Some(now);
   507→                    a.triggered_px = Some(mid.normalize().to_string());
   508→                    fired.push(a.clone());
   509→                }
   510→            }
   511→        }
   512→        for a in &fired {
   513→            tracing::info!(
   514→                "[alerts] #{} {} {} {} crossed (mid {})",
   515→                a.id,
   516→                a.market.label(),
   517→                a.dir.label(),
   518→                a.price,
   519→                a.triggered_px.as_deref().unwrap_or("?")
   520→            );
   521→        }
   522→        fired
   523→    }

   527→    /// Build the full wire event for one market (fully owned, cheap to hold).
   528→    fn build_event(&self, market: Market) -> Option<WireEvent> {
   529→        let now = now_ms();
   530→        let books = self.books.lock().unwrap();
   531→        let mb = books.get(&market)?;
   532→
   533→        let venue_book = |st: &Option<VenueState>, feed: &VenueFeed| -> Option<VenueBook> {
   534→            let st = st.as_ref()?;
   535→            Some(VenueBook {
   536→                health: VenueHealth {
   537→                    status: feed.status(),
   538→                    msg_per_sec: feed.rate(),
   539→                    last_msg_age_ms: now.saturating_sub(feed.last_msg_ms()),
   540→                    levels: st.total_levels(),
   541→                },
   542→                bids: st.top_bids(WIRE_DEPTH),
   543→                asks: st.top_asks(WIRE_DEPTH),
   544→            })
   545→        };
   546→
   547→        let consolidated = match (mb.hl.as_ref(), mb.lt.as_ref()) {
   548→            (Some(h), Some(l)) => {
   549→                let (bids, asks) = consolidate(h, l, WIRE_DEPTH);
   550→                Some(ConsolidatedBook { bids, asks })
   551→            }
   552→            _ => None,
   553→        };
   554→
   555→        let stats: BookStats = compute_stats(mb.hl.as_ref(), mb.lt.as_ref());
   556→
   557→        Some(WireEvent::Book {
   558→            market,
   559→            hyperliquid: venue_book(&mb.hl, &self.hl),
   560→            lighter: venue_book(&mb.lt, &self.lt),
   561→            consolidated,
   562→            stats,
   563→            ts: now,
   564→        })
   565→    }

   568→        serde_json::to_string(&self.build_event(market)?).ok()
   569→    }

   571→    /// Compact ticker JSON for one market (nil when no book yet).
   572→    pub fn ticker_json(&self, market: Market) -> Option<String> {

   575→            let mb = books.get(&market)?;
   576→            compute_stats(mb.hl.as_ref(), mb.lt.as_ref())
   577→        };
   578→        let ev = WireEvent::Ticker {
   579→            market,
   580→            mid: stats.mid,
   581→            spread_bps: stats.spread_bps,

   584→        };
   585→        serde_json::to_string(&ev).ok()
   586→    }

   588→    /// JSON payload for GET /api/health.
   589→    pub fn health_payload(&self) -> serde_json::Value {
   590→        let now = now_ms();
   591→        let feed = |f: &VenueFeed| {
   592→            json!({
   593→                "status": f.status().label(),
   594→                "msgs_per_sec": (f.rate() * 10.0).round() / 10.0,
   595→                "total_msgs": f.total(),
   596→                "last_msg_age_ms": now.saturating_sub(f.last_msg_ms()),
   597→            })
   598→        };
   599→        let books = self.books.lock().unwrap();
   600→        let tapes = self.tapes.lock().unwrap();
   601→        let history = self.history.lock().unwrap();
   602→        let selections = self.selections.lock().unwrap();
   603→        let markets: Vec<serde_json::Value> = Market::ALL
   604→            .iter()
   605→            .map(|m| {
   606→                let mb = books.get(m);
   607→                json!({
   608→                    "market": m.label(),
   609→                    "hyperliquid_levels": mb.and_then(|b| b.hl.as_ref()).map(|s| s.total_levels()).unwrap_or(0),
   610→                    "lighter_levels": mb.and_then(|b| b.lt.as_ref()).map(|s| s.total_levels()).unwrap_or(0),
   611→                    "tape_trades": tapes.get(m).map(|t| t.trades.len()).unwrap_or(0),
   612→                    "history_samples": history.get(m).map(|q| q.len()).unwrap_or(0),
   613→                    "viewers": selections.get(m).copied().unwrap_or(0),
   614→                })
   615→            })
   616→            .collect();
   617→        json!({
   618→            "uptime_ms": now.saturating_sub(self.started_ms),
   619→            "hyperliquid": feed(&self.hl),
   620→            "lighter": feed(&self.lt),
   621→            "active_alerts": self.alerts.lock().unwrap().iter().filter(|a| a.triggered_ms.is_none()).count(),
   622→            "markets": markets,
   623→        })
   624→    }

   626→    /// Coalescing publisher:
   627→    /// - 10 Hz: full `Book` for selected markets, `Trades` batches, alert checks
   628→    /// - 2 Hz:  compact `Ticker` for every dirty market
   629→    /// - 0.5 Hz: `Status` heartbeat
   630→    pub async fn publish_loop(self: Arc<Self>) {
   631→        let mut tick100 =
   632→            tokio::time::interval(std::time::Duration::from_millis(100));
   633→        tick100.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
   634→        let mut tick500 =
   635→            tokio::time::interval(std::time::Duration::from_millis(500));
   636→        tick500.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
   637→        let mut hl_s = 0u64;
   638→        let mut hl_t = now_ms();
   639→        let mut lt_s = 0u64;
   640→        let mut lt_t = hl_t;
   641→        let mut heartbeat = 0u32;
   642→        // (mid, spread_bps, imbalance) waiting for the 2 Hz ticker flush.
   643→        let mut ticker_accum: HashMap<Market, (String, String, String)> = HashMap::new();
   644→
   645→        loop {
   646→            tokio::select! {
   647→                _ = tick100.tick() => {
   648→                    // EMA bookkeeping (10 Hz samples).
   649→                    self.hl.tick_rate(&mut hl_s, &mut hl_t);
   650→                    self.lt.tick_rate(&mut lt_s, &mut lt_t);
   651→
   652→                    // 1) Dirty books -> full Book events (selected markets only)
   653→                    //    + accumulate tickers for everyone.
   654→                    let mut built: Vec<(Market, WireEvent)> = Vec::with_capacity(4);
   655→                    {
   656→                        let mut books = self.books.lock().unwrap();
   657→                        let selections = self.selections.lock().unwrap();
   658→                        for (m, mb) in books.iter_mut() {
   659→                            if !mb.dirty {
   660→                                continue;
   661→                            }
   662→                            mb.dirty = false;
   663→                            let stats = compute_stats(mb.hl.as_ref(), mb.lt.as_ref());
   664→                            ticker_accum.insert(
   665→                                *m,
   666→                                (stats.mid.clone(), stats.spread_bps.clone(), stats.imbalance.clone()),
   667→                            );
   668→                            if selections.get(m).copied().unwrap_or(0) > 0 {
   669→                                if let Some(ev) = build_book_event(*m, mb, &stats, &self.hl, &self.lt) {
   670→                                    built.push((*m, ev));
   671→                                }
   672→                            }
   673→                        }
   674→                    }
   675→                    for (_, ev) in built {
   676→                        self.broadcast(BcKind::Book, market_of(&ev), &ev);
   677→                    }
   678→
   679→                    // 2) Flush trade batches.
   680→                    for (market, trades) in self.take_pending() {
   681→                        self.broadcast(BcKind::Trades, market, &WireEvent::Trades { market, trades });
   682→                    }
   683→
   684→                    // 3) Alert engine.
   685→                    for a in self.check_alerts() {
   686→                        self.broadcast(BcKind::AlertFired, a.market, &WireEvent::AlertFired { alert: a });
   687→                    }
   688→                }
   689→                _ = tick500.tick() => {
   690→                    // Ticker flush.
   691→                    if !ticker_accum.is_empty() {
   692→                        let now = now_ms();
   693→                        let entries: Vec<(Market, (String, String, String))> =
   694→                            ticker_accum.drain().collect();
   695→                        for (market, (mid, spread_bps, imbalance)) in entries {
   696→                            let ev = WireEvent::Ticker { market, mid, spread_bps, imbalance, ts: now };
   697→                            self.broadcast(BcKind::Ticker, market, &ev);
   698→                        }
   699→                    }
   700→
   701→                    // Heartbeat.
   702→                    heartbeat += 1;
   703→                    if heartbeat % 4 == 0 {
   704→                        let st = WireEvent::Status {
   705→                            hyperliquid: self.hl.status(),
   706→                            lighter: self.lt.status(),
   707→                            ts: now_ms(),
   708→                        };
   709→                        self.broadcast(BcKind::Status, Market::Eth, &st);
   710→                    }
   711→                }
   712→            }
   713→        }
   714→    }
   715→
   716→    /// History sampler loop (spawned by main).
   717→    pub async fn history_loop(self: Arc<Self>) {
   718→        let mut tick = tokio::time::interval(std::time::Duration::from_secs(SAMPLE_SECS));
   719→        tick.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
   720→        loop {
   721→            tick.tick().await;
   722→            self.sample_all();
   723→        }
   724→    }
   725→}

   727→/// Extract the market tag from a Book wire event (for routing).
   728→fn market_of(ev: &WireEvent) -> Market {
   729→    match ev {
   730→        WireEvent::Book { market, .. } => *market,
   731→        WireEvent::Ticker { market, .. }
   732→        | WireEvent::Trades { market, .. }
   733→        | WireEvent::History { market, .. } => *market,
   734→        WireEvent::AlertFired { alert } => alert.market,
   735→        _ => Market::Eth,
   736→    }
   737→}

   739→/// Build the full Book event from an already-locked MarketBook + precomputed
   740→/// stats (shares the health snapshot with the caller).
   741→fn build_book_event(
   742→    market: Market,
   743→    mb: &MarketBook,
   744→    stats: &BookStats,
   745→    hl_feed: &VenueFeed,
   746→    lt_feed: &VenueFeed,
   747→) -> Option<WireEvent> {
   748→    let now = now_ms();
   749→    let venue_book = |st: &Option<VenueState>, feed: &VenueFeed| -> Option<VenueBook> {
   750→        let st = st.as_ref()?;
   751→        Some(VenueBook {
   752→            health: VenueHealth {
   753→                status: feed.status(),
   754→                msg_per_sec: feed.rate(),
   755→                last_msg_age_ms: now.saturating_sub(feed.last_msg_ms()),
   756→                levels: st.total_levels(),
   757→            },
   758→            bids: st.top_bids(WIRE_DEPTH),
   759→            asks: st.top_asks(WIRE_DEPTH),
   760→        })
   761→    };
   762→    let consolidated = match (mb.hl.as_ref(), mb.lt.as_ref()) {
   763→        (Some(h), Some(l)) => {
   764→            let (bids, asks) = consolidate(h, l, WIRE_DEPTH);
   765→            Some(ConsolidatedBook { bids, asks })
   766→        }
   767→        _ => None,
   768→    };
   769→    Some(WireEvent::Book {
   770→        market,
   771→        hyperliquid: venue_book(&mb.hl, hl_feed),
   772→        lighter: venue_book(&mb.lt, lt_feed),
   773→        consolidated,
   774→        stats: stats.clone(),
   775→        ts: now,
   776→    })
   777→}

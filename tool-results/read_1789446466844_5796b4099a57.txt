     1→     1→//! Central book registry: shared state between venue connectors, the publish
     2→     2→//! loop, the depth sampler and the HTTP/WS layer.
     3→     3→//!
     4→     4→//! Scaling model (12 markets): every socket selects ONE market. The broadcast
     5→     5→//! channel carries a compact `Broadcast` envelope — (kind, market, json) — so
     6→     6→//! each socket can route events without re-parsing JSON:
     7→     7→//! - `Book`   → forwarded only to sockets whose selection matches (full depth)
     8→     8→//! - `Ticker` → forwarded to everyone (compact, coalesced at 2 Hz)
     9→     9→//! - `Trades` → forwarded to everyone (batches, small)
    10→    10→//! - `AlertSet` / `AlertFired` / `Status` → forwarded to everyone
    11→    11→

    26→    26→pub fn now_ms() -> u64 {
    27→    27→    SystemTime::now()
    28→    28→        .duration_since(UNIX_EPOCH)
    29→    29→        .map(|d| d.as_millis() as u64)
    30→    30→        .unwrap_or(0)
    31→    31→}

   100→   100→
   101→   101→    /// Track the message rate with a light EMA. Called at 10 Hz.
   102→   102→    fn tick_rate(&self, last_sample: &mut u64, last_ts: &mut u64) -> f32 {
   103→   103→        let now = now_ms();
   104→   104→        let total = self.total();
   105→   105→        let dt = (now - *last_ts).max(1) as f32 / 1000.0;
   106→   106→        let inst = (total.saturating_sub(*last_sample)) as f32 / dt;
   107→   107→        *last_sample = total;
   108→   108→        *last_ts = now;
   109→   109→        let mut ema = self.ema_rate.lock().unwrap();
   110→   110→        *ema = *ema * 0.85 + inst * 0.15;
   111→   111→        *ema
   112→   112→    }
   113→   113→
   114→   114→    pub fn rate(&self) -> f32 {
   115→   115→        *self.ema_rate.lock().unwrap()
   116→   116→    }

   263→   263→    /// Push venue trades into the tape (dedup by venue id, keep a backlog).
   264→   264→    pub fn push_trades(&self, market: Market, trades: Vec<Trade>) {
   265→   265→        let mut tapes = self.tapes.lock().unwrap();
   266→   266→        let st = tapes.entry(market).or_default();
   267→   267→        for t in trades {
   268→   268→            if st.seen.contains(&t.id) {
   269→   269→                continue;
   270→   270→            }
   271→   271→            st.seen.insert(t.id.clone());
   272→   272→            st.seen_order.push_back(t.id.clone());
   273→   273→            // Bound the dedup set to ~2x the backlog (evict in arrival order).
   274→   274→            while st.seen_order.len() > 2 * TAPE_BACKLOG {
   275→   275→                if let Some(old) = st.seen_order.pop_front() {
   276→   276→                    st.seen.remove(&old);
   277→   277→                }
   278→   278→            }
   279→   279→            st.trades.push_back(t.clone());
   280→   280→            while st.trades.len() > TAPE_BACKLOG {
   281→   281→                st.trades.pop_front();
   282→   282→            }
   283→   283→            st.pending.push(t);
   284→   284→        }
   285→   285→    }

   287→   287→    /// Drain trades pending publication (called by the publish loop at 10 Hz).
   288→   288→    fn take_pending(&self) -> Vec<(Market, Vec<Trade>)> {
   289→   289→        let mut tapes = self.tapes.lock().unwrap();
   290→   290→        tapes
   291→   291→            .iter_mut()
   292→   292→            .filter(|(_, st)| !st.pending.is_empty())
   293→   293→            .map(|(m, st)| (*m, std::mem::take(&mut st.pending)))
   294→   294→            .collect()
   295→   295→    }

   309→   309→    /// Sample depth bands for every market with live book data. Called every
   310→   310→    /// `SAMPLE_SECS` by a dedicated task; keeps a `SAMPLE_WINDOW` ring buffer.
   311→   311→    pub fn sample_all(&self) {
   312→   312→        let now = now_ms();
   313→   313→        // Band thresholds: 0.1%, 0.5%, 1%, 2%.
   314→   314→        let ths = [
   315→   315→            Decimal::new(1, 3),
   316→   316→            Decimal::new(5, 3),
   317→   317→            Decimal::new(1, 2),
   318→   318→            Decimal::new(2, 2),
   319→   319→        ];
   320→   320→        let mut samples: Vec<(Market, DepthSample)> = Vec::new();
   321→   321→        {
   322→   322→            let books = self.books.lock().unwrap();
   323→   323→            for (m, mb) in books.iter() {
   324→   324→                let (hl, lt) = (mb.hl.as_ref(), mb.lt.as_ref());
   325→   325→                if hl.is_none() && lt.is_none() {
   326→   326→                    continue;
   327→   327→                }
   328→   328→                let Some(mid) = cross_mid(hl, lt) else { continue };
   329→   329→
   330→   330→                let mut bands = [Decimal::ZERO; 4];
   331→   331→                let mut asks = [Decimal::ZERO; 4];
   332→   332→                for st in [hl, lt].into_iter().flatten() {
   333→   333→                    let b = st.band_notionals(mid, &ths, ob_core::Side::Bid);
   334→   334→                    let a = st.band_notionals(mid, &ths, ob_core::Side::Ask);
   335→   335→                    for i in 0..4 {
   336→   336→                        bands[i] += b[i];
   337→   337→                        asks[i] += a[i];
   338→   338→                    }
   339→   339→                }
   340→   340→
   341→   341→                // Spread over the cross-venue touch.
   342→   342→                let bb = [hl.and_then(|b| b.best_bid()), lt.and_then(|b| b.best_bid())]
   343→   343→                    .into_iter()
   344→   344→                    .flatten()
   345→   345→                    .max_by(|x, y| x.0.cmp(&y.0));
   346→   346→                let ba = [hl.and_then(|b| b.best_ask()), lt.and_then(|b| b.best_ask())]
   347→   347→                    .into_iter()
   348→   348→                    .flatten()
   349→   349→                    .min_by(|x, y| x.0.cmp(&y.0));
   350→   350→                let sb = match (bb, ba) {
   351→   351→                    (Some(bb), Some(ba)) => {
   352→   352→                        if mid.is_zero() {
   353→   353→                            Decimal::ZERO
   354→   354→                        } else {
   355→   355→                            (ba.0 - bb.0) / mid * Decimal::from(10_000u64)
   356→   356→                        }
   357→   357→                    }
   358→   358→                    _ => Decimal::ZERO,
   359→   359→                };
   360→   360→
   361→   361→                let total = bands[1] + asks[1];
   362→   362→                let imb = if total.is_zero() {
   363→   363→                    Decimal::ZERO
   364→   364→                } else {
   365→   365→                    (bands[1] - asks[1]) / total
   366→   366→                };
   367→   367→
   368→   368→                let f = |d: Decimal| d.to_f64().unwrap_or(0.0);
   369→   369→                samples.push((
   370→   370→                    *m,
   371→   371→                    DepthSample {
   372→   372→                        t: now,
   373→   373→                        mid: f(mid),
   374→   374→                        sb: f(sb.round_dp(2)),
   375→   375→                        b1: f(bands[0]),
   376→   376→                        a1: f(asks[0]),
   377→   377→                        b5: f(bands[1]),
   378→   378→                        a5: f(asks[1]),
   379→   379→                        b10: f(bands[2]),
   380→   380→                        a10: f(asks[2]),
   381→   381→                        b20: f(bands[3]),
   382→   382→                        a20: f(asks[3]),
   383→   383→                        im: f(imb.round_dp(4)),
   384→   384→                    },
   385→   385→                ));
   386→   386→            }
   387→   387→        }
   388→   388→        if samples.is_empty() {
   389→   389→            return;
   390→   390→        }
   391→   391→        let mut hist = self.history.lock().unwrap();
   392→   392→        for (m, s) in samples {
   393→   393→            let q = hist.entry(m).or_default();
   394→   394→            // Skip duplicate timestamps (same sample window).
   395→   395→            if q.back().map(|b| b.t == s.t).unwrap_or(false) {
   396→   396→                continue;
   397→   397→            }
   398→   398→            q.push_back(s);
   399→   399→            while q.len() > SAMPLE_WINDOW {
   400→   400→                q.pop_front();
   401→   401→            }
   402→   402→        }
   403→   403→    }
   404→   404→

   427→   427→    /// Create an alert (validates the price parses as a positive decimal).
   428→   428→    pub fn create_alert(&self, market: Market, dir: AlertDir, price: String) -> Result<Alert, String> {
   429→   429→        let th: Decimal = price
   430→   430→            .parse()
   431→   431→            .map_err(|_| format!("price {price:?} is not a valid decimal"))?;
   432→   432→        if !th.is_sign_positive() {
   433→   433→            return Err("price must be positive".into());
   434→   434→        }
   435→   435→        let alert = Alert {
   436→   436→            id: self.next_alert_id.fetch_add(1, Ordering::Relaxed),
   437→   437→            market,
   438→   438→            dir,
   439→   439→            price: th.normalize().to_string(),
   440→   440→            created_ms: now_ms(),
   441→   441→            triggered_ms: None,
   442→   442→            triggered_px: None,
   443→   443→        };
   444→   444→        let mut alerts = self.alerts.lock().unwrap();
   445→   445→        alerts.push(alert.clone());
   446→   446→        drop(alerts);
   447→   447→        self.broadcast(
   448→   448→            BcKind::AlertSet,
   449→   449→            alert.market,
   450→   450→            &WireEvent::AlertSet {
   451→   451→                alerts: self.alerts_snapshot(),
   452→   452→            },
   453→   453→        );
   454→   454→        Ok(alert)
   455→   455→    }

   457→   457→    pub fn delete_alert(&self, id: u64) -> bool {
   458→   458→        let removed = {
   459→   459→            let mut alerts = self.alerts.lock().unwrap();
   460→   460→            let before = alerts.len();
   461→   461→            alerts.retain(|a| a.id != id);
   462→   462→            before != alerts.len()
   463→   463→        };
   464→   464→        if removed {
   465→   465→            self.broadcast(
   466→   466→                BcKind::AlertSet,
   467→   467→                Market::Eth,
   468→   468→                &WireEvent::AlertSet {
   469→   469→                    alerts: self.alerts_snapshot(),
   470→   470→                },
   471→   471→            );
   472→   472→        }
   473→   473→        removed
   474→   474→    }

   480→   480→    /// Check every active alert against the live cross-venue mid. Returns the
   481→   481→    /// alerts that fired during this pass (already mutated in place).
   482→   482→    fn check_alerts(&self) -> Vec<Alert> {
   483→   483→        let now = now_ms();
   484→   484→        let mut fired = Vec::new();
   485→   485→        {
   486→   486→            let books = self.books.lock().unwrap();
   487→   487→            let mut alerts = self.alerts.lock().unwrap();
   488→   488→            for a in alerts.iter_mut() {
   489→   489→                if a.triggered_ms.is_some() {
   490→   490→                    continue;
   491→   491→                }
   492→   492→                let Ok(th) = a.price.parse::<Decimal>() else {
   493→   493→                    continue;
   494→   494→                };
   495→   495→                let Some(mb) = books.get(&a.market) else {
   496→   496→                    continue;
   497→   497→                };
   498→   498→                let Some(mid) = cross_mid(mb.hl.as_ref(), mb.lt.as_ref()) else {
   499→   499→                    continue;
   500→   500→                };
   501→   501→                let hit = match a.dir {
   502→   502→                    AlertDir::Above => mid >= th,
   503→   503→                    AlertDir::Below => mid <= th,
   504→   504→                };
   505→   505→                if hit {
   506→   506→                    a.triggered_ms = Some(now);
   507→   507→                    a.triggered_px = Some(mid.normalize().to_string());
   508→   508→                    fired.push(a.clone());
   509→   509→                }
   510→   510→            }
   511→   511→        }
   512→   512→        for a in &fired {
   513→   513→            tracing::info!(
   514→   514→                "[alerts] #{} {} {} {} crossed (mid {})",
   515→   515→                a.id,
   516→   516→                a.market.label(),
   517→   517→                a.dir.label(),
   518→   518→                a.price,
   519→   519→                a.triggered_px.as_deref().unwrap_or("?")
   520→   520→            );
   521→   521→        }
   522→   522→        fired
   523→   523→    }

   527→   527→    /// Build the full wire event for one market (fully owned, cheap to hold).
   528→   528→    fn build_event(&self, market: Market) -> Option<WireEvent> {
   529→   529→        let now = now_ms();
   530→   530→        let books = self.books.lock().unwrap();
   531→   531→        let mb = books.get(&market)?;
   532→   532→
   533→   533→        let venue_book = |st: &Option<VenueState>, feed: &VenueFeed| -> Option<VenueBook> {
   534→   534→            let st = st.as_ref()?;
   535→   535→            Some(VenueBook {
   536→   536→                health: VenueHealth {
   537→   537→                    status: feed.status(),
   538→   538→                    msg_per_sec: feed.rate(),
   539→   539→                    last_msg_age_ms: now.saturating_sub(feed.last_msg_ms()),
   540→   540→                    levels: st.total_levels(),
   541→   541→                },
   542→   542→                bids: st.top_bids(WIRE_DEPTH),
   543→   543→                asks: st.top_asks(WIRE_DEPTH),
   544→   544→            })
   545→   545→        };
   546→   546→
   547→   547→        let consolidated = match (mb.hl.as_ref(), mb.lt.as_ref()) {
   548→   548→            (Some(h), Some(l)) => {
   549→   549→                let (bids, asks) = consolidate(h, l, WIRE_DEPTH);
   550→   550→                Some(ConsolidatedBook { bids, asks })
   551→   551→            }
   552→   552→            _ => None,
   553→   553→        };
   554→   554→
   555→   555→        let stats: BookStats = compute_stats(mb.hl.as_ref(), mb.lt.as_ref());
   556→   556→
   557→   557→        Some(WireEvent::Book {
   558→   558→            market,
   559→   559→            hyperliquid: venue_book(&mb.hl, &self.hl),
   560→   560→            lighter: venue_book(&mb.lt, &self.lt),
   561→   561→            consolidated,
   562→   562→            stats,
   563→   563→            ts: now,
   564→   564→        })
   565→   565→    }

   603→   603→        let markets: Vec<serde_json::Value> = Market::ALL
   604→   604→            .iter()
   605→   605→            .map(|m| {

   607→   607→                json!({
   608→   608→                    "market": m.label(),
   609→   609→                    "hyperliquid_levels": mb.and_then(|b| b.hl.as_ref()).map(|s| s.total_levels()).unwrap_or(0),
   610→   610→                    "lighter_levels": mb.and_then(|b| b.lt.as_ref()).map(|s| s.total_levels()).unwrap_or(0),
   611→   611→                    "tape_trades": tapes.get(m).map(|t| t.trades.len()).unwrap_or(0),
   612→   612→                    "history_samples": history.get(m).map(|q| q.len()).unwrap_or(0),

   622→   622→            "markets": markets,
   623→   623→        })

   626→   626→    /// Coalescing publisher:
   627→   627→    /// - 10 Hz: full `Book` for selected markets, `Trades` batches, alert checks
   628→   628→    /// - 2 Hz:  compact `Ticker` for every dirty market
   629→   629→    /// - 0.5 Hz: `Status` heartbeat
   630→   630→    pub async fn publish_loop(self: Arc<Self>) {
   631→   631→        let mut tick100 =
   632→   632→            tokio::time::interval(std::time::Duration::from_millis(100));
   633→   633→        tick100.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
   634→   634→        let mut tick500 =
   635→   635→            tokio::time::interval(std::time::Duration::from_millis(500));
   636→   636→        tick500.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);
   637→   637→        let mut hl_s = 0u64;
   638→   638→        let mut hl_t = now_ms();
   639→   639→        let mut lt_s = 0u64;
   640→   640→        let mut lt_t = hl_t;
   641→   641→        let mut heartbeat = 0u32;
   642→   642→        // (mid, spread_bps, imbalance) waiting for the 2 Hz ticker flush.
   643→   643→        let mut ticker_accum: HashMap<Market, (String, String, String)> = HashMap::new();
   644→   644→
   645→   645→        loop {
   646→   646→            tokio::select! {
   647→   647→                _ = tick100.tick() => {
   648→   648→                    // EMA bookkeeping (10 Hz samples).
   649→   649→                    self.hl.tick_rate(&mut hl_s, &mut hl_t);
   650→   650→                    self.lt.tick_rate(&mut lt_s, &mut lt_t);
   651→   651→
   652→   652→                    // 1) Dirty books -> full Book events (selected markets only)
   653→   653→                    //    + accumulate tickers for everyone.
   654→   654→                    let mut built: Vec<(Market, WireEvent)> = Vec::with_capacity(4);
   655→   655→                    {
   656→   656→                        let mut books = self.books.lock().unwrap();
   657→   657→                        let selections = self.selections.lock().unwrap();
   658→   658→                        for (m, mb) in books.iter_mut() {
   659→   659→                            if !mb.dirty {
   660→   660→                                continue;
   661→   661→                            }
   662→   662→                            mb.dirty = false;
   663→   663→                            let stats = compute_stats(mb.hl.as_ref(), mb.lt.as_ref());
   664→   664→                            ticker_accum.insert(
   665→   665→                                *m,
   666→   666→                                (stats.mid.clone(), stats.spread_bps.clone(), stats.imbalance.clone()),
   667→   667→                            );
   668→   668→                            if selections.get(m).copied().unwrap_or(0) > 0 {
   669→   669→                                if let Some(ev) = build_book_event(*m, mb, &stats, &self.hl, &self.lt) {
   670→   670→                                    built.push((*m, ev));
   671→   671→                                }
   672→   672→                            }
   673→   673→                        }
   674→   674→                    }
   675→   675→                    for (_, ev) in built {
   676→   676→                        self.broadcast(BcKind::Book, market_of(&ev), &ev);
   677→   677→                    }
   678→   678→
   679→   679→                    // 2) Flush trade batches.
   680→   680→                    for (market, trades) in self.take_pending() {
   681→   681→                        self.broadcast(BcKind::Trades, market, &WireEvent::Trades { market, trades });
   682→   682→                    }
   683→   683→
   684→   684→                    // 3) Alert engine.
   685→   685→                    for a in self.check_alerts() {
   686→   686→                        self.broadcast(BcKind::AlertFired, a.market, &WireEvent::AlertFired { alert: a });
   687→   687→                    }
   688→   688→                }
   689→   689→                _ = tick500.tick() => {
   690→   690→                    // Ticker flush.
   691→   691→                    if !ticker_accum.is_empty() {
   692→   692→                        let now = now_ms();
   693→   693→                        let entries: Vec<(Market, (String, String, String))> =
   694→   694→                            ticker_accum.drain().collect();
   695→   695→                        for (market, (mid, spread_bps, imbalance)) in entries {
   696→   696→                            let ev = WireEvent::Ticker { market, mid, spread_bps, imbalance, ts: now };
   697→   697→                            self.broadcast(BcKind::Ticker, market, &ev);
   698→   698→                        }
   699→   699→                    }
   700→   700→
   701→   701→                    // Heartbeat.
   702→   702→                    heartbeat += 1;
   703→   703→                    if heartbeat % 4 == 0 {
   704→   704→                        let st = WireEvent::Status {
   705→   705→                            hyperliquid: self.hl.status(),
   706→   706→                            lighter: self.lt.status(),
   707→   707→                            ts: now_ms(),
   708→   708→                        };
   709→   709→                        self.broadcast(BcKind::Status, Market::Eth, &st);
   710→   710→                    }
   711→   711→                }
   712→   712→            }
   713→   713→        }
   714→   714→    }

   716→   716→    /// History sampler loop (spawned by main).

   718→   718→        let mut tick = tokio::time::interval(std::time::Duration::from_secs(SAMPLE_SECS));
   719→   719→        tick.set_missed_tick_behavior(tokio::time::MissedTickBehavior::Skip);

   722→   722→            self.sample_all();
   723→   723→        }
   724→   724→    }
